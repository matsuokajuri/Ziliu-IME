cmake_minimum_required(VERSION 3.25)
project(ZiliuScorerChecks LANGUAGES CXX)

# Focused portable checks, not a non-Windows build of the Windows application.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ZILIU_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ZILIU_SOURCE_ROOT}/cmake/ZiliuWarnings.cmake")
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
find_package(Threads REQUIRED)
add_library(ziliu_core STATIC "${ZILIU_SOURCE_ROOT}/src/core/src/scorer.cpp")
target_include_directories(ziliu_core PUBLIC "${ZILIU_SOURCE_ROOT}/src/core/include")
target_link_libraries(ziliu_core PUBLIC Threads::Threads)
ziliu_configure_target(ziliu_core)
include(CTest)
enable_testing()

add_executable(ziliu_scorer_tests "${ZILIU_SOURCE_ROOT}/tests/scorer_tests.cpp")
target_link_libraries(ziliu_scorer_tests PRIVATE ziliu_core)
ziliu_configure_target(ziliu_scorer_tests)
add_test(NAME ziliu_scorer_tests COMMAND ziliu_scorer_tests)

add_executable(ziliu_onnx_graph_gate_tests "${ZILIU_SOURCE_ROOT}/tests/onnx_graph_gate_tests.cpp"
  "${ZILIU_SOURCE_ROOT}/src/broker/src/onnx_graph_gate.cpp")
target_include_directories(ziliu_onnx_graph_gate_tests PRIVATE "${ZILIU_SOURCE_ROOT}/src/broker/src")
ziliu_configure_target(ziliu_onnx_graph_gate_tests)
add_test(NAME ziliu_onnx_graph_gate_tests COMMAND ziliu_onnx_graph_gate_tests)
set_tests_properties(ziliu_scorer_tests ziliu_onnx_graph_gate_tests PROPERTIES TIMEOUT 30)

option(ZILIU_ENABLE_ONNX_SCORER "Compile optional ORT adapter with an externally supplied header" OFF)
if(ZILIU_ENABLE_ONNX_SCORER)
  include("${ZILIU_SOURCE_ROOT}/cmake/ZiliuScorer.cmake")
  ziliu_add_onnx_scorer()
  add_executable(ziliu_scorer_asset_tests "${ZILIU_SOURCE_ROOT}/tests/scorer_asset_tests.cpp")
  target_include_directories(ziliu_scorer_asset_tests PRIVATE "${ZILIU_SOURCE_ROOT}/src/broker/src")
  target_link_libraries(ziliu_scorer_asset_tests PRIVATE ziliu_onnx_scorer)
  ziliu_configure_target(ziliu_scorer_asset_tests)
  add_test(NAME ziliu_scorer_asset_tests COMMAND ziliu_scorer_asset_tests)
  set_tests_properties(ziliu_scorer_asset_tests PROPERTIES TIMEOUT 30)
endif()
